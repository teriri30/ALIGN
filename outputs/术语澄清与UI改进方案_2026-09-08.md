## 📖 **第一部分：术语澄清（核心概念）**

### **1. 种植模式的正确理解**

#### **物理种植布局**（用户视角）
```
田块横截面示意图（垂直于行向）：

|←2行种植带→|←宽行→|←  4行种植带  →|←宽行→|←2行种植带→|←宽行→|←  4行种植带  →|...
   ○ ○        空     ○ ○ ○ ○        空     ○ ○        空     ○ ○ ○ ○
  (边缘)             (主体，窄行间距)        (边缘)             (主体)

说明：
- ○ 代表一行水稻
- 空白区域 = 宽行（约60-80cm，用于履带行走）
- 4行之间的间距 = 窄行（约20-30cm，株间通风，不能碾压）
- 2行种植带位于边缘，是过渡区
```

#### **收割机几何参数**
- **收割机幅宽（cutter_width_m）**：1.8m
- **幅宽容纳范围**：`1行(边缘) + 宽行 + 4行主体 + 宽行 + 1行(边缘)` = 刚好1.8m
- **一次收割效果**：
  - 收掉完整的 4行种植带
  - 同时收掉左右两个2行种植带的各一半（各1行）
  - 相邻两趟作业会把中间的2行种植带完全收完

#### **术语定义对照表**

| 专业术语（代码用） | 用户友好表述（UI显示） | 具体含义 |
|-------------------|---------------------|---------|
| 4行种植带（4-row band） | 主体作业区 / 4行带 | 4行水稻，窄行间距，是主要收割目标 |
| 2行种植带（2-row band） | 边缘过渡区 / 2行带 | 2行水稻，位于4行带边缘，被相邻两趟自然收完 |
| 宽行（wide row） | 履带通道 / 宽行 | 60-80cm宽的空隙，履带行走区域 |
| 窄行（narrow row） | 株间空隙 / 窄行 | 4行带内部的3个20-30cm间距 |
| 作业带（work band） | 收割作业带 | 软件识别出的可作业区域，对应4行种植带 |
| 中心线（centerline） | 作业线 / 收割路线 | 收割机行走的中心路径 |
| 幅宽（cutter width） | 收割幅宽 | 收割机一次能收的宽度（1.8m） |
| 割幅（effective width） | 有效割幅 | 实际作业宽度 |

#### **3. 错误提示翻译（技术术语→通俗语言）**

```python
# 在 footprint_planner.py 中新增翻译字典
METRIC_TRANSLATIONS = {
    # 指标名称翻译
    "track_outside_field_pct": "履带越出田界比例",
    "track_forbidden_overlap_pct": "履带进入禁行区比例",
    "track_core_overlap_pct": "履带碾压作物核心区比例",
    "track_outside_support_pct": "履带在无作物支撑区比例",
    "track_uncertain_overlap_pct": "履带在不确定区域比例",
    "planned_target_geometric_coverage_pct": "割刀覆盖率",
    "rolling_canopy_pct": "预计碾压率",
    "crop_mask_geometric_coverage_pct": "作物掩膜覆盖率",
    
    # 单位翻译
    "pct": "%",
    "m": "米",
    "m2": "平方米",
    
    # 状态翻译
    "exceeds limit": "超过限制",
    "below minimum": "低于最低要求",
}

# 用户友好的错误描述生成函数
def translate_validation_issue(tech_description: str) -> str:
    """将技术描述翻译为用户友好的错误提示"""
    
    # 示例：
    # 输入："track_outside_field_pct=5.2 exceeds limit=2.0"
    # 输出："❌ 履带越出田界 5.2%（限制：2.0%）→ 请调整路径，避免越界"
    
    result = tech_description
    
    # 1. 替换技术术语
    for tech, user in METRIC_TRANSLATIONS.items():
        result = result.replace(tech, user)
    
    # 2. 美化格式
    if "=" in result:
        # "履带越出田界比例=5.2 超过限制=2.0"
        # → "履带越出田界 5.2%（限制：2.0%）"
        parts = result.split()
        if len(parts) >= 2:
            metric_part = parts[0].replace("=", " ")  # "履带越出田界比例 5.2"
            limit_part = parts[-1].replace("=", "：")  # "超过限制：2.0"
            result = f"{metric_part}（{limit_part}）"
    
    # 3. 添加图标和建议
    suggestions = {
        "履带越出田界": "→ 请调整路径，或扩大田块圈选范围",
        "履带进入禁行区": "→ 请拖动作业线避开禁行区",
        "割刀覆盖率": "→ 覆盖率略低，可调整作业线位置",
        "预计碾压率": "→ 碾压率略高，可优化转弯策略",
    }
    
    icon = "❌ " if "超过限制" in result or "低于" in result else "⚠️ "
    result = icon + result
    
    for keyword, suggestion in suggestions.items():
        if keyword in result:
            result += f"\n   {suggestion}"
    
    return result
```

```python
def _flash_centerline_on_canvas(self, centerline_id: str, duration_ms: int = 2000):
    """在画布上闪烁高亮某条中心线"""
    # 找到对应的中心线几何
    centerline = self._get_centerline_geometry(centerline_id)

        return
    
    # 绘制加粗高亮版本
    self._highlight_centerline = centerline_id
    self._draw_overlay()
    
    # 2秒后恢复正常
    QTimer.singleShot(duration_ms, lambda: self._clear_highlight())

    """清除高亮"""
    self._highlight_centerline = None
    self._draw_overlay()
```

```python
def _on_rollback_task(self, task_id: str):
    """回退到指定任务之前"""
    label = TASK_LABELS.get(task_id, task_id)
    
    # 简化的确认对话框
    msg_box = QMessageBox(self)
    msg_box.setIcon(QMessageBox.Question)
    msg_box.setWindowTitle("确认回退")
    msg_box.setText(f"将回退到"{label}"之前，后续结果会清除")
    msg_box.setInformativeText("可通过"编辑→撤销"恢复（若未关闭软件）")
    
    # 增加"不再提示"选项
    chk_dont_ask = QCheckBox("下次直接回退（不再确认）")
    msg_box.setCheckBox(chk_dont_ask)
    
    msg_box.setStandardButtons(QMessageBox.Yes | QMessageBox.No)
    msg_box.setDefaultButton(QMessageBox.No)
    
    reply = msg_box.exec()
    
    if chk_dont_ask.isChecked():
        self.cfg.set("rollback_no_confirm", True)
    
    if reply == QMessageBox.Yes:
        self._do_rollback(task_id)
```
