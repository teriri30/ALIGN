```python
# 在 import 区域后增加
METRIC_TRANSLATIONS = {
    # 指标名称翻译
    "track_outside_field_pct": "履带越出田界比例",
    "track_forbidden_overlap_pct": "履带进入禁行区比例",
    "track_core_overlap_pct": "履带碾压作物核心区比例",
    "track_outside_support_pct": "履带在无作物支撑区比例",
    "track_uncertain_overlap_pct": "履带在不确定区域比例",
    "planned_target_geometric_coverage_pct": "割刀覆盖率",
    "rolling_canopy_pct": "预计碾压率",
    "crop_mask_geometric_coverage_pct": "作物掩膜覆盖率",
    "machine_outside_field_pct": "机身越出田界比例",
    "machine_forbidden_overlap_pct": "机身进入禁行区比例",
    
    # 符号翻译
    "exceeds limit": "超过限制",
    "below minimum": "低于最低要求",
}

ISSUE_SUGGESTIONS = {
    "履带越出田界": "→ 请调整路径，或扩大田块圈选范围",
    "履带进入禁行区": "→ 请拖动作业线避开禁行区",
    "割刀覆盖率": "→ 覆盖率略低，可调整作业线位置",
    "预计碾压率": "→ 碾压率略高，可优化转弯策略或作业线位置",
    "机身越出田界": "→ 请调整路径，确保机身完全在田界内",
}

def translate_validation_issue(description: str, severity: str = "error") -> str:
    """将技术指标描述翻译为用户友好提示
    
    示例：
    输入："track_outside_field_pct=5.2 exceeds limit=2.0"
    输出："❌ 履带越出田界比例 5.2%（限制：2.0%）\n   → 请调整路径，或扩大田块圈选范围"
    """
    result = str(description)
    
    # 1. 替换技术术语
    for tech, user in METRIC_TRANSLATIONS.items():
        result = result.replace(tech, user)
    
    # 2. 美化数值格式
    if "=" in result:
        # "履带越出田界比例=5.2 超过限制=2.0"
        # → "履带越出田界比例 5.2%（限制：2.0%）"
        parts = result.split()
        if len(parts) >= 2:
            # 提取指标部分
            metric_part = parts[0].replace("=", " ")  # "履带越出田界比例 5.2"
            # 提取限制部分
            if "超过限制=" in result:
                limit_value = result.split("超过限制=")[-1].strip()
                result = f"{metric_part}（限制：{limit_value}）"
            elif "低于最低要求=" in result:
                limit_value = result.split("低于最低要求=")[-1].strip()
                result = f"{metric_part}（最低要求：{limit_value}）"
    
    # 3. 添加图标
    icon = "❌" if severity == "error" else "⚠️"
    result = f"{icon} {result}"
    
    # 4. 添加改进建议（仅对 warning）
    if severity == "warning":
        for keyword, suggestion in ISSUE_SUGGESTIONS.items():
            if keyword in result:
                result += f"\n   {suggestion}"
                break
    
    return result
```

## ✅ **任务4：掩膜处理结果增加说明文字（0.5小时）**

### **文件：src/pyside6_app/main_window.py**

#### **改动4.1：_on_mask_process_done 函数增加详细日志**
```python
def _on_mask_process_done(self, payload):
    # ... 原有逻辑 ...
    
    # 新增：统计并说明识别结果
    mask_result = self.state.mask_result
    if mask_result:
        wide_bands = mask_result.get("wide_bands") or []
        all_bands = mask_result.get("all_bands") or []
        
        wide_count = len(wide_bands)
        narrow_count = len([b for b in all_bands if b.get("band_type") == "narrow"])
        
        # 详细说明
        message = f"✅ 掩膜处理完成\n"
        message += f"   识别结果：{wide_count} 条4行主体区"
        if narrow_count > 0:
            message += f" + {narrow_count} 条2行边缘区"
        
        message += (
            f"\n   说明：4行主体区用于生成作业线，2行边缘区会被相邻两趟自然收完"
            f"\n   （收割机幅宽1.8m = 左1行+宽行+4行+宽行+右1行）"
            f"\n   接下来点击"生成作业线""
        )
        
        self.log_panel.info(message)
        self.lbl_status.setText(f"掩膜处理完成（{wide_count}条主体区）")
    else:
        # 原有简单提示
        self.log_panel.info("掩膜处理完成")
        self.lbl_status.setText("掩膜处理完成")
```
